  doser/.style={
    font=\fontsize{6.5}{7.5}\selectfont,

    ymin=0.5, ymax=6.5,
    ytick={1,...,6},
    y tick style={draw=none},
    axis line style={muted, line width=0.6pt},
    tick style={ink2, line width=0.5pt},

    scatter/classes={
      research-tap-in={mark=*, draw=res, fill=res},

      surrogate-tap-in={mark=*, draw=surr, fill=surr},

      surrogate-fine-out={mark=triangle, draw=surr, fill=white}},
    every mark/.append style={line width=0.45pt, scale=0.62, fill opacity=0.85},
  },
}

% Row shading, alternate rows, as in Fig. 7.
\newcommand{\rowbands}[2]{%

  \fill[band] (axis cs:#1,3.5) rectangle (axis cs:#2,4.5);
  \fill[band] (axis cs:#1,5.5) rectangle (axis cs:#2,6.5);}
